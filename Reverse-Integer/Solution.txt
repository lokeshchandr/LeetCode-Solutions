class Solution {
public:
    int reverse(int x) {
        int reverse = 0;
        while (x != 0) {
            int dig = x % 10;
            x = x / 10;
            if (reverse > INT_MAX / 10 ||
                (reverse == INT_MAX / 10 && dig > 7)) {
                return 0; // overflow case
            }
            if (reverse < INT_MIN / 10 ||
                (reverse == INT_MIN / 10 && dig < -8)) {
                return 0; // underflow case
            }
            reverse = reverse * 10 + dig;
        }
        return reverse;
    }
};
